### harness.py

```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable

@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float
    max_ms: float

def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):
        h = h ^ b
        h = (h * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h

def run_benchmark(fn: Callable[[], None], warmup_runs: int = 3, timed_runs: int = 9) -> BenchStats:
    for _ in range(warmup_runs):
        fn()
    samples = []
    for _ in range(timed_runs):
        t0 = time.perf_counter_ns()
        fn()
        t1 = time.perf_counter_ns()
        samples.append((t1 - t0) / 1e6)
    
    samples.sort()
    mid = timed_runs // 2
    med = samples[mid] if timed_runs % 2 == 1 else 0.5 * (samples[mid - 1] + samples[mid])
    q25 = samples[timed_runs // 4]
    q75 = samples[(3 * timed_runs) // 4]
    
    return BenchStats(samples[0], med, q75 - q25, samples[-1])

def emit_verify_tag(task_id: str, student_seed: int, primary_metric: float, secondary_metric: float) -> str:
    hw = mp.cpu_count()
    payload = f"{task_id}|{student_seed}|{hw}|{primary_metric:.4f}|{secondary_metric:.4f}"
    sig = fnv1a_hash(payload)
    return f"VERIFY_HASH [{task_id}]: {sig:016X} (P_log={hw})"
```

```python
import multiprocessing as mp
from multiprocessing import shared_memory
import ctypes
from harness import run_benchmark, emit_verify_tag

def fs_worker(shm_name: str, slot_idx: int, iters: int) -> None:
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    
    acc = arr[slot_idx]
    mask = 0xFFFFFFFFFFFFFFFF
    mult = 6364136223846793005
    
    for i in range(iters):
        acc = (acc * mult + (i + 1)) & mask
        arr[slot_idx] = acc  # Direct hardware store to shared L1 cache line
        
    del arr
    shm.close()

def run_parallel_fs(p: int, stride: int, shm_name: str, total_iters: int):
    iters_per_worker = total_iters // p
    processes = []

        slot_idx = t * stride
        proc = mp.Process(target=fs_worker, args=(shm_name, slot_idx, iters_per_worker))
        processes.append(proc)
        proc.start()
        
    for proc in processes:
        proc.join()
```

```python
import multiprocessing as mp
import math
import time
from harness import run_benchmark, emit_verify_tag

def stage_a(n_elements: int) -> float:
    """Stage A: Strictly Serial Reduction (12 transcendental ops per element)"""
    acc = 0.0
    for i in range(n_elements):
        acc += math.sin(i) + math.sqrt(i)
        acc = math.sin(acc) + math.sqrt(abs(acc))
        acc = math.sin(acc) + math.sqrt(abs(acc))
    return acc

def stage_b_worker(chunk_size: int) -> float:
    """Stage B: Embarrassingly Parallel Compute (35 FLOPs/trig per element)"""
    acc = 0.0
    for i in range(chunk_size):
        v = i * 0.1
        for _ in range(7):
            v = v * 1.5 + math.sin(v)
        acc += v
    return acc

def execute_pipeline(pool, p: int, N: int):
    # Stage A: First 5% of elements executed in parent
    n_a = int(0.05 * N)
    stage_a(n_a)
    
    # Stage B: Remaining elements partitioned evenly across pool
    base_chunk = N // p
    chunks = [base_chunk] * p
    chunks[-1] += N % p  # add any remainder to the last worker
    
    pool.map(stage_b_worker, chunks)

def measure_stage_a(N: int, timed_runs: int = 5) -> float:
    n_a = int(0.05 * N)
    samples = []
    for _ in range(timed_runs):
        t0 = time.perf_counter_ns()
        stage_a(n_a)
        t1 = time.perf_counter_ns()
        samples.append((t1 - t0) / 1e6)
    samples.sort()
    return samples[timed_runs // 2]

if __name__ == "__main__":
    P_PHYS = 8
    P_LOG = 12
    N_STRONG = 1_600_000
    N0_WEAK = 250_000
    
    print(f"--- CPU Detected: P_phys={P_PHYS}, P_log={P_LOG} ---")
    
    # --- Experiment 2A: Strong Scaling Sweep ---
    print("\n--- Experiment 2A: Strong Scaling (N = 1.6M) ---")
    T_1_strong = None
    strong_results = {}
    
    for p in range(1, P_LOG + 1):
        with mp.Pool(processes=p) as pool:
            bench_fn = lambda pool=pool, p=p: execute_pipeline(pool, p, N_STRONG)
            stats = run_benchmark(bench_fn, warmup_runs=1, timed_runs=5)
            T_p = stats.median_ms
            
            if p == 1:
                T_1_strong = T_p
            
            S_p = T_1_strong / T_p
            E_p = S_p / p
            strong_results[p] = S_p
            print(f"p={p:2d} | T_p={T_p:7.2f} ms | S_p={S_p:5.2f}x | E_p={E_p:4.2f}")
            
    # --- Experiment 2B: Weak Scaling Sweep ---
    print("\n--- Experiment 2B: Weak Scaling (N_0 = 250k/worker) ---")
    T_1_weak = None
    gustafson_results = {}
    
    for p in range(1, P_LOG + 1):
        N_weak = p * N0_WEAK
        with mp.Pool(processes=p) as pool:
            bench_fn = lambda pool=pool, p=p, N=N_weak: execute_pipeline(pool, p, N)
```
